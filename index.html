function pilihMood() { let mood = document.getElementById("mood").value if (mood === "senang") { document.body.className = "" document.getElementById("teks").innerHTML = "Hari ini kamu senang 😏🤤" document.body.classList.add("senang") } else if (mood === "sedih") { document.body.className = "" document.getElementById("teks").innerHTML = "Hari ini kamu sedih 😭💔" document.body.classList.add("sedih") } else if (mood === "marah") { document.body.className = "" document.getElementById("teks").innerHTML ="Hari ini kamu marah 🤬🤬" document.body.classList.add("marah") } else if (mood === "semangat") { document.body.className = "" document.getElementById("teks").innerHTML = "Hari ini kamu semangat 💪😈" document.body.classList.add("semangat") } else if (mood === "default") { document.getElementById("teks").innerHTML = "Pilih mood dulu..." document.body.className = "" document.body.classList.add("default") } } mood.addEventListener("change", pilihMood)
